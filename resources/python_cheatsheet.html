class Persona:
    """Representa una persona."""

    def __init__(self, nombre: str, edad: int):
        self.nombre = nombre
        self.edad = edad

    def saludar(self) -> str:
        return f"Hola, soy {self.nombre}"

    def __str__(self) -> str:
        return f"Persona({self.nombre}, {self.edad})"

    def __repr__(self) -> str:
        return f"Persona(nombre='{self.nombre}', edad={self.edad})"

p = Persona("Ana", 30)
print(p.saludar())  # Hola, soy Ana
print(p)            # Persona(Ana, 30) -> __str__

@property

class Circulo:
    def __init__(self, radio: float):
        self._radio = radio

    @property
    def area(self) -> float:
        return 3.14159 * self._radio ** 2

    @property
    def radio(self) -> float:
        return self._radio

    @radio.setter
    def radio(self, valor: float):
        if valor < 0:
            raise ValueError("El radio no puede ser negativo")
        self._radio = valor

c = Circulo(5)
c.area    # 78.54 (como atributo, sin paréntesis)
c.radio = 10  # usa el setter

from functools import wraps

def timer(func):
    import time
    @wraps(func)
    def wrapper(*args, **kwargs):
        inicio = time.perf_counter()
        resultado = func(*args, **kwargs)
        fin = time.perf_counter()
        print(f"{func.__name__} tardó {fin - inicio:.4f}s")
        return resultado
    return wrapper

@timer
def procesar(data):
    # ... proceso ...
    pass

Decorador con argumentos

def repetir(veces: int):
    def decorador(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for _ in range(veces):
                resultado = func(*args, **kwargs)
            return resultado
        return wrapper
    return decorador

@repetir(veces=3)
def gritar():
    print("¡AHHH!")

import re

# Buscar patrón
texto = "Mi email es ana@example.com y mi tlf es 555-1234"

# search: primera coincidencia
m = re.search(r"\b\d{3}-\d{4}\b", texto)
if m:
    print(m.group())  # 555-1234

# findall: todas las coincidencias
emails = re.findall(r"\b[\w.]+@[\w.]+\b", texto)
# ['ana@example.com']

# sub: reemplazar
limpio = re.sub(r"\d", "#", texto)
# "Mi email es ana@example.com y mi tlf es ###-####"

# split: dividir por patrón
partes = re.split(r"[,;\s]+", "a, b; c d")
# ['a', 'b', 'c', 'd']

# Grupos nombrados
m = re.search(r"(?P\w+)@(?P\w+\.\w+)", texto)
m.group("usuario")   # ana
m.group("dominio")   # example.com

# Compilar patrón (reutilizar)
patron = re.compile(r"^\d{4}-\d{2}-\d{2}$")
patron.match("2025-01-15")  # Match object

Async / Await avanzado

import asyncio

async def tarea(nombre: str, segundos: float):
    print(f"{nombre} inicia")
    await asyncio.sleep(segundos)
    print(f"{nombre} termina ({segundos}s)")
    return nombre

# Ejecutar una coroutine
async def main():
    # Concurrente: ambas corren a la vez
    resultados = await asyncio.gather(
        tarea("A", 2),
        tarea("B", 1),
    )
    print(resultados)  # ['A', 'B'] (en orden de llamada)

asyncio.run(main())
# A inicia / B inicia / B termina (1s) / A termina (2s)

asyncio.create_task

async def main():
    # Crear tarea (se ejecuta en background)
    t = asyncio.create_task(tarea("C", 3))

    # Hacer otras cosas...
    await asyncio.sleep(1)
    print("Haciendo algo...")

    # Esperar a que termine
    resultado = await t
    print(f"Resultado: {resultado}")

def calcular_total(precio: float, cantidad: int, descuento: float = 0) -> float:
    """Calcula el total con descuento aplicado.

    Args:
        precio: Precio unitario del producto.
        cantidad: Número de unidades.
        descuento: Porcentaje de descuento (0-100). Por defecto 0.

    Returns:
        El total calculado tras aplicar el descuento.

    Raises:
        ValueError: Si el descuento no está entre 0 y 100.
    """
    if not 0 <= descuento <= 100:
        raise ValueError("Descuento debe estar entre 0 y 100")
    subtotal = precio * cantidad
    return subtotal * (1 - descuento / 100)

from contextlib import contextmanager

@contextmanager
def temporizador():
    import time
    inicio = time.perf_counter()
    try:
        yield  # código del bloque with
    finally:
        fin = time.perf_counter()
        print(f"Tardó {fin - inicio:.4f}s")
